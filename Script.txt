-- ============================================================
-- HELPERS

local function make(class, props, parent)
    local inst = Instance.new(class)
    for k, v in pairs(props) do inst[k] = v end
    if parent then inst.Parent = parent end
    return inst
end

local function corner(radius, parent)
    return make("UICorner", { CornerRadius = UDim.new(0, radius) }, parent)
end

local function stroke(color, thickness, transparency, parent)
    return make("UIStroke", { Color = color, Thickness = thickness or 1, Transparency = transparency or 0 }, parent)
end

local function gradient(c1, c2, rotation, parent)
    return make("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, c1),
            ColorSequenceKeypoint.new(1, c2),
        }),
        Rotation = rotation or 0,
    }, parent)
end

local function triGradient(c1, c2, c3, rotation, parent)
    return make("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, c1),
            ColorSequenceKeypoint.new(0.5, c2),
            ColorSequenceKeypoint.new(1, c3),
        }),
        Rotation = rotation or 0,
    }, parent)
end

local function padding(top, bottom, left, right, parent)
    return make("UIPadding", {
        PaddingTop = UDim.new(0, top or 0),
        PaddingBottom = UDim.new(0, bottom or 0),
        PaddingLeft = UDim.new(0, left or 0),
        PaddingRight = UDim.new(0, right or 0),
    }, parent)
end

local function listLayout(props, parent)
    return make("UIListLayout", props, parent)
end

-- Smooth color sequence helper (avoids stale reference issues when tweening)
local function seq(c1, c2)
    return ColorSequence.new({
        ColorSequenceKeypoint.new(0, c1),
        ColorSequenceKeypoint.new(1, c2),
    })
end

-- ============================================================
-- TOAST HELPER
-- ============================================================
local toastTimer = nil
local function showToast(message, isError)
    toastLabel.Text = message or "Copied!"
    local iconColor = isError and C.danger or C.success
    toastIconHolder.BackgroundColor3 = iconColor
    toastIcon.Text = isError and "!" or "✓"
    toastStroke.Color = isError and C.danger or C.accent

    if toastTimer then task.cancel(toastTimer); toastTimer = nil end

    toast.Visible = true
    toast.Position = UDim2.new(0.5, 0, 1, -30)
    toast.BackgroundTransparency = 0.05

    TweenService:Create(toast, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Position = UDim2.new(0.5, 0, 1, -60),
    }):Play()
    TweenService:Create(toastStroke, TweenInfo.new(0.3), { Transparency = 0.4 }):Play()

    toastTimer = task.delay(2.2, function()
        local fade = TweenService:Create(toast, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
            Position = UDim2.new(0.5, 0, 1, -30),
            BackgroundTransparency = 1,
        })
        fade:Play()
        TweenService:Create(toastStroke, TweenInfo.new(0.25), { Transparency = 1 }):Play()
        fade.Completed:Wait()
        toast.Visible = false
    end)
end

-- ============================================================
-- COPY HELPER
-- ============================================================
local function copyToClipboard(text, message)
    if not text or text == "" then showToast("Nothing to copy", true); return end
    pcall(function()
        if setclipboard then setclipboard(text)
        elseif toclipboard then toclipboard(text) end
    end)
    showToast(message or "Copied!", false)
end

-- ============================================================
-- LIVE ANIMATION SYSTEM
-- ============================================================
local t = 0
local animConn
animConn = RunService.Heartbeat:Connect(function(dt)
    t = t + dt

    -- 1. Shimmer sweeps
    local st = (t * 0.6) % 1
    copyShimmer.Position = UDim2.new(st * 1.4 - 0.2, 0, -0.1, 0)
    continueShimmer.Position = UDim2.new((st * 1.4 - 0.2) % 1.4 - 0.2, 0, -0.1, 0)

    -- 2. Card surface sweep (slow)
    local cs = (t * 0.18) % 1
    cardSweep.Position = UDim2.new(cs * 1.6 - 0.3, 0, -0.1, 0)

    -- 3. Icon ring pulse
    iconRing.Transparency = 0.4 + math.sin(t * 2.5) * 0.15
    iconHaloStroke.Transparency = 0.85 + math.sin(t * 2.5 + 1) * 0.08

    -- 4. Card glow breathing
    cardGlowStroke.Transparency = 0.85 + math.sin(t * 1.8) * 0.05
    cardGlow.ImageTransparency = 0.5 + math.sin(t * 1.2) * 0.08

    -- 5. Title gradient sweep
    titleGrad.Offset = Vector2.new(math.sin(t * 0.8) * 0.3, 0)

    -- 6. Status dot pulse + glow
    if statusDot.BackgroundColor3 == C.accentHi then
        local s = 0.85 + math.sin(t * 3) * 0.15
        statusDot.Size = UDim2.fromScale(s, s)
        statusGlow.BackgroundTransparency = 0.9 - math.sin(t * 3) * 0.08
    end

    -- 7. Orbs drift
    local d1 = math.sin(t * 0.4) * 14
    local d2 = math.cos(t * 0.35) * 10
    orb1.Position = UDim2.new(0, -140 + d1, 0, -160 + d2)
    orb2.Position = UDim2.new(1, 110 - d1, 1, 130 + d2)
    orb3.Position = UDim2.new(1, 100 + d2, 0, -40 + d1 * 0.5)
    orb4.Position = UDim2.new(0, -90 + d2, 1, 100 + d1 * 0.5)

    -- 8. Sparkles drift + twinkle
    for i, s in ipairs(sparkles) do
        local f = s.frame
        local p = f.Position
        local newY = p.Y.Scale + s.speed * dt
        if newY > 1.05 then newY = -0.05 end
        local newX = p.X.Scale + math.sin(t * 0.6 + s.phase) * s.amp * dt
        f.Position = UDim2.fromScale(newX, newY)
        local twinkle = 0.65 + math.sin(t * 2 + s.phase) * 0.25
        f.BackgroundTransparency = twinkle
    end

    -- 9. Backdrop gradient rotation (very slow)
    backdropGrad.Rotation = 135 + math.sin(t * 0.15) * 8
end)
